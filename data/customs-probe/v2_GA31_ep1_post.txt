	    /*
	     * Replace all SVG images with inline SVG
	     */
	    $('img.svg').each(function(){
	        var $img = jQuery(this);
	        var imgID = $img.attr('id');
	        var imgClass = $img.attr('class');
	        var imgURL = $img.attr('src');

	        jQuery.get(imgURL, function(data) {
	            // Get the SVG tag, ignore the rest
	            var $svg = jQuery(data).find('svg');

	            // Add replaced image's ID to the new SVG
	            if(typeof imgID !== 'undefined') {
	                $svg = $svg.attr('id', imgID);
	            }
	            // Add replaced image's classes to the new SVG
	            if(typeof imgClass !== 'undefined') {
	                $svg = $svg.attr('class', imgClass+' replaced-svg');
	            }

	            // Remove any invalid XML tags as per http://validator.w3.org
	            $svg = $svg.removeAttr('xmlns:a');

	            // Check if the viewport is set, if the viewport is not set the SVG wont't scale.
	            if(!$svg.attr('viewBox') && $svg.attr('height') && $svg.attr('width')) {
	                $svg.attr('viewBox', '0 0 ' + $svg.attr('height') + ' ' + $svg.attr('width'))
	            }

	            // Replace image with new SVG
	            $img.replaceWith($svg);

	        }, 'xml');

	    });
	});

$( document ).ready(function(){
	var pid='item_'+window.location.pathname.replace(/%3b/, ";").split("/").pop().split(";")[0].split("_")[0];

});
function changeAriaEpanded(item){
    if($('#'+item).attr("aria-expanded")=="true"){
        $('.accordion_multilayer').attr("aria-expanded","false");
    	$('.accordion_multilayer ul').attr("style","overflow: hidden; display: none;");
    }
    else{
        $('.accordion li').removeClass("active");
    	$('.accordion_multilayer').attr("aria-expanded","false");
    	$('.accordion_multilayer ul').attr("style","overflow: hidden; display: none;");
    	
    	$('#'+item).addClass("active");
    	$('#'+item).attr("aria-expanded","true");
    	$('#'+item+' ul').attr("style","overflow: hidden; display: block;");
    }
}

function numberFormatter(cellvalue, options, rowObject){
    if(cellvalue!="-" && cellvalue!="D"){
        return Math.round(Number(cellvalue)).toLocaleString('en-US');
    }
    else{
        return cellvalue;
    }
}

jQuery(document).ready(function() {
    if($("#sysMsg").length>0){
        $("#blocker").hide();
    }
    
 	//jqGrid colNames
 	var columnList = new DOMParser().parseFromString('[]', 'text/html').documentElement.textContent.slice(1, -1).split(", ");

 	    var element = columnList[i];
 	    
 	    if(element!="貨品號列" && element!="數量(限11碼貨品)"){
 	       columnNames.push(element);
 	    }
 	    else if(element=="貨品號列"){
 	       columnNames.push("貨品號列");
 	       columnNames.push("中文貨名");
 	       columnNames.push("英文貨名");
 	    }
 	    else if(element=="數量(限11碼貨品)"){
 	       columnNames.push("數量");
 	       columnNames.push("數量單位");
 	    }
 	}

 	var columnModelGroup = new Array();
 	
 	tdIdx=1;

 	    var element = columnNames[i];
 	 	var columnModel = new Array();
 	    
 	    columnModel["name"]=element;
 	    columnModel["index"]="td"+tdIdx;
 	    columnModel["sortable"]=false;
 	    
 	    if(element=="進出口別"){
 	        columnModel["width"]="58";
 	 	    columnModel["align"]="center";
 	    }

   	           columnModel["width"]="70";

 	 	    columnModel["align"]="center";
 	 	    columnModel["classes"]="warp";
 	    }
 	    else if(element=="數量單位"){
 	        columnModel["width"]="40";
 	 	    columnModel["align"]="center";
 	    }
 	    else if(element=="貨品號列"){
 	        if($("#GOODS_TYPE").val()=="1"){
 	 	       columnModel["width"]="35";
 	           columnModel["align"]="center";
 	        }
 	        else{
  	 	       columnModel["width"]="55";
 	           columnModel["align"]="left";
 	        }
 	    }
 	    else if(element=="國家"){
 	        columnModel["width"]="40";
 	 	    columnModel["align"]="left";
 	    }
 	    else if(element=="中文貨名" || element=="英文貨名"){
 	 	    columnModel["align"]="left";
 	    }
 	    else if(element=="美元(千元)" || element=="新臺幣(千元)" || element=="重量(公噸)" || element=="重量(公斤)"){
 	        columnModel["width"]="55";
 	 	    columnModel["align"]="right";
 	 	    columnModel["formatter"]=numberFormatter;
 	    }
 	    else if(element=="數量"){
 	        columnModel["width"]="45";
 	 	    columnModel["align"]="right";
 	 	    columnModel["formatter"]=numberFormatter;
 	    }
 	    columnModelGroup.push(columnModel);
 	    tdIdx++;
 	}
